% Make API endpoint paths breakable (Appendix B). Neither \nolinkurl (hyperref)
% nor an xstring \StrSubstitute reliably broke the longest paths, because the
% "[id]"/"[skillId]" brackets defeat url-style breaking and xstring's optional
% argument scanning. \allowbreak after "/" and "]" alone also left the longest
% path (#7, .../skills/[skillId]/verification) overflowing into the next column.
% This self-contained tokenizer walks the path one token at a time.
\makeatletter
\newcommand{\apiep}[1]{\begingroup\ttfamily\apiep@loop#1\@nil}
% Walk the path one token at a time. A real \discretionary break is inserted
% after every character (so a path can ALWAYS wrap and never overflow its
% column), while "/" and "]" additionally get a zero-penalty break so TeX
% prefers to wrap at those semantic boundaries when it can.
\def\apiep@loop#1{%
  \ifx#1\@nil
    \endgroup
  \else
    #1%
    \if#1/\penalty0 \else\if#1]\penalty0 \else\discretionary{}{}{}\fi\fi
    \expandafter\apiep@loop
  \fi}
\makeatother
